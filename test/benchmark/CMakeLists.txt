# Solver micro-benchmark
#
# Standalone benchmark that runs one mechanism at large grid-cell counts so
# per-Solve() work dominates over per-call overhead. Used to check for
# performance regressions in the linear solver and related hot paths.
#
# Mechanisms: chapman (7 reactions) and ts1 (547 reactions). Run
# `micm_bench list` to print every valid mechanism and solver combination.
#
# Not registered as a ctest. Run directly (see scripts/bench_micm.sh) or
# under callgrind (see scripts/profile_micm.sh).
#
# MICM_ENABLE_BENCHMARK gates this directory and defaults to OFF.

add_executable(micm_bench micm_bench.cpp)
target_link_libraries(micm_bench PRIVATE musica::micm)
# Debug info lets callgrind_annotate attribute counts to source lines.
target_compile_options(micm_bench PRIVATE $<$<COMPILE_LANG_AND_ID:CXX,GNU,Clang,AppleClang,NVHPC>:-g>)
set_target_properties(micm_bench PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")

# ts1_mechanism.hpp is generated by import_ts1.py and checked in, so the build
# needs no extra input. Re-run that script when the mechanism changes.

if(NOT ${MICM_GPU_TYPE} STREQUAL "None")
  target_link_libraries(micm_bench PRIVATE micm_cuda)
endif()
